```python
import re
from bisect import bisect_left, bisect_right
from operator import itemgetter

_DIGITS_RE = re.compile(r'[0-9]+')
_EVENTS = frozenset(('OPEN', 'PRIORITY', 'PAUSE', 'RESUME', 'CLOSE', 'REOPEN', 'HOLIDAY'))
_LIMITS = {'P1': 240, 'P2': 480, 'P3': 1440, 'P4': 2400}

_NOT_OPENED = 0
_RUNNING = 1
_PAUSED = 2
_CLOSED = 3
_STATUS_NAME = {_RUNNING: 'running', _PAUSED: 'paused', _CLOSED: 'closed'}


def _to_int(s):
    s = s.lstrip('0') or '0'
    try:
        return int(s)
    except ValueError:
        v = 0
        for i in range(0, len(s), 1000):
            chunk = s[i:i + 1000]
            v = v * (10 ** len(chunk)) + int(chunk)
        return v


def compute_sla(stream):
    hol = set()
    events = []

    for raw in stream:
        if isinstance(raw, (bytes, bytearray)):
            try:
                raw = raw.decode('utf-8')
            except Exception:
                continue
        if not isinstance(raw, str):
            continue
        parts = raw.strip().split(',')
        np = len(parts)
        if np != 3 and np != 4:
            continue
        f1 = parts[0].strip()
        tid = parts[1].strip()
        ev = parts[2].strip()
        if not f1 or _DIGITS_RE.fullmatch(f1) is None:
            continue
        if not tid:
            continue
        if ev not in _EVENTS:
            continue
        if ev == 'OPEN' or ev == 'PRIORITY':
            if np != 4:
                continue
            p = parts[3].strip()
            if p not in _LIMITS:
                continue
        else:
            if np != 3:
                continue
            p = None
        if ev == 'HOLIDAY':
            if tid != '*':
                continue
        elif tid == '*':
            continue
        m = _to_int(f1)
        if ev == 'HOLIDAY':
            d = m // 1440
            q, r = divmod(d, 7)
            if r < 5:
                hol.add(q * 5 + r)
            continue
        events.append((m, tid, ev, p))

    if not events:
        return []

    now = max(e[0] for e in events)
    events.sort(key=itemgetter(0))  # stable

    per = {}
    for e in events:
        lst = per.get(e[1])
        if lst is None:
            per[e[1]] = [e]
        else:
            lst.append(e)

    H = sorted(hol)
    G = [h - i for i, h in enumerate(H)]
    nH = len(H)

    def bmin(t):
        # number of business minutes in [0, t)
        d, mod = divmod(t, 1440)
        q, r = divmod(d, 7)
        if r < 5:
            W = q * 5 + r
            hc = bisect_left(H, W)
            res = (W - hc) * 480
            if mod > 540 and not (hc < nH and H[hc] == W):
                res += 480 if mod >= 1020 else mod - 540
        else:
            W = q * 5 + 5
            hc = bisect_left(H, W)
            res = (W - hc) * 480
        return res

    def nth(n):
        # minute index of the n-th (0-indexed) business minute
        D, off = divmod(n, 480)
        i = bisect_right(G, D)
        j = D + i
        q, r = divmod(j, 5)
        return (q * 7 + r) * 1440 + 540 + off

    results = []
    for tid in sorted(per):
        evs = per[tid]
        state = _NOT_OPENED
        prio = None
        used = 0
        last = 0
        br = None
        opened = False
        i = 0
        n = len(evs)
        while i < n:
            m = evs[i][0]
            if state == _RUNNING and m > last:
                bl = bmin(last)
                bm = bmin(m)
                if br is None:
                    k = _LIMITS[prio] + 1 - used
                    if k < 1:
                        k = 1
                    if bm - bl >= k:
                        t = nth(bl + k - 1) + 1
                        if t < m:
                            br = t
                used += bm - bl
            last = m
            while i < n and evs[i][0] == m:
                _, _, ev, p = evs[i]
                i += 1
                if ev == 'OPEN':
                    if state == _NOT_OPENED or state == _CLOSED:
                        state = _RUNNING
                        prio = p
                        used = 0
                        br = None
                        opened = True
                elif ev == 'PRIORITY':
                    if state == _RUNNING or state == _PAUSED:
                        prio = p
                elif ev == 'PAUSE':
                    if state == _RUNNING:
                        state = _PAUSED
                elif ev == 'RESUME':
                    if state == _PAUSED:
                        state = _RUNNING
                elif ev == 'CLOSE':
                    if state == _RUNNING or state == _PAUSED:
                        state = _CLOSED
                elif ev == 'REOPEN':
                    if state == _CLOSED:
                        state = _RUNNING
            if state != _NOT_OPENED and br is None and used > _LIMITS[prio]:
                br = m

        if not opened:
            continue

        if state == _RUNNING and now > last:
            bl = bmin(last)
            bn = bmin(now)
            if br is None:
                k = _LIMITS[prio] + 1 - used
                if k < 1:
                    k = 1
                if bn - bl >= k:
                    t = nth(bl + k - 1) + 1
                    if t <= now:
                        br = t
            used += bn - bl
            last = now
        if br is None and used > _LIMITS[prio]:
            br = now

        results.append({
            "ticket_id": tid,
            "priority": prio,
            "used_minutes": used,
            "breached": br is not None,
            "breached_at": br,
            "status": _STATUS_NAME[state],
        })

    return results
```

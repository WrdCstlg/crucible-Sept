from bisect import bisect_left
from operator import itemgetter

_EVENTS = frozenset(("OPEN", "PRIORITY", "PAUSE", "RESUME", "CLOSE", "REOPEN", "HOLIDAY"))
_LIMITS = {"P1": 240, "P2": 480, "P3": 1440, "P4": 2400}

_NOT_OPENED = 0
_RUNNING = 1
_PAUSED = 2
_CLOSED = 3

_STATUS_NAMES = {_RUNNING: "running", _PAUSED: "paused", _CLOSED: "closed"}


def _parse_int(s):
    s2 = s.lstrip("0")
    if not s2:
        return 0
    try:
        return int(s2)
    except ValueError:
        # Extremely long digit strings (int max str digits limit); convert in chunks.
        val = 0
        for i in range(0, len(s2), 1000):
            chunk = s2[i:i + 1000]
            val = val * (10 ** len(chunk)) + int(chunk)
        return val


def _weekday_business(t):
    """Number of weekday business minutes (ignoring holidays) in [0, t)."""
    w, r = divmod(t, 10080)
    d, m = divmod(r, 1440)
    if d >= 5:
        return w * 2400 + 2400
    m -= 540
    if m < 0:
        m = 0
    elif m > 480:
        m = 480
    return w * 2400 + d * 480 + m


def compute_sla(stream):
    holidays = set()
    events = []

    for line in stream:
        if not isinstance(line, str):
            continue
        parts = line.strip().split(",")
        n = len(parts)
        if n != 3 and n != 4:
            continue
        ms = parts[0].strip()
        if not ms or not ms.isascii() or not ms.isdigit():
            continue
        tid = parts[1].strip()
        if not tid:
            continue
        ev = parts[2].strip()
        if ev not in _EVENTS:
            continue
        if ev == "OPEN" or ev == "PRIORITY":
            if n != 4:
                continue
            p = parts[3].strip()
            if p not in _LIMITS:
                continue
        else:
            if n != 3:
                continue
            p = None
        if ev == "HOLIDAY":
            if tid != "*":
                continue
            holidays.add(_parse_int(ms) // 1440)
            continue
        if tid == "*":
            continue
        events.append((_parse_int(ms), tid, ev, p))

    if not events:
        return []

    events.sort(key=itemgetter(0))  # stable
    now = events[-1][0]

    # Holiday calendar (only weekday holidays matter)
    hd = sorted(d for d in holidays if d % 7 < 5)
    nh = len(hd)
    hb = [_weekday_business(d * 1440) - 480 * i for i, d in enumerate(hd)]

    W = _weekday_business

    def B(t):
        """Business minutes in [0, t)."""
        res = W(t)
        d, m = divmod(t, 1440)
        k = bisect_left(hd, d)
        res -= 480 * k
        if k < nh and hd[k] == d:
            m -= 540
            if m < 0:
                m = 0
            elif m > 480:
                m = 480
            res -= m
        return res

    def Binv(K):
        """Smallest t with B(t) >= K (K >= 1)."""
        j = bisect_left(hb, K)
        k0 = K + 480 * j - 1
        w, r = divmod(k0, 2400)
        d, m = divmod(r, 480)
        return w * 10080 + d * 1440 + 540 + m + 1

    per = {}
    for t, tid, ev, p in events:
        lst = per.get(tid)
        if lst is None:
            lst = []
            per[tid] = lst
        lst.append((t, ev, p))

    result = []
    for tid in sorted(per):
        evs = per[tid]
        state = _NOT_OPENED
        prio = None
        used = 0
        b_last = 0
        breached = None
        i = 0
        L = len(evs)
        while i < L:
            t = evs[i][0]
            bt = B(t)
            if state == _RUNNING:
                new_used = used + bt - b_last
                if breached is None:
                    lim = _LIMITS[prio]
                    if new_used > lim:
                        tb = Binv(lim - used + b_last + 1)
                        if tb < t:
                            breached = tb
                used = new_used
            b_last = bt
            # Apply all events at minute t
            while i < L and evs[i][0] == t:
                _, ev, p = evs[i]
                i += 1
                if ev == "OPEN":
                    if state == _NOT_OPENED or state == _CLOSED:
                        state = _RUNNING
                        prio = p
                        used = 0
                        breached = None
                elif ev == "PRIORITY":
                    if state == _RUNNING or state == _PAUSED:
                        prio = p
                elif ev == "PAUSE":
                    if state == _RUNNING:
                        state = _PAUSED
                elif ev == "RESUME":
                    if state == _PAUSED:
                        state = _RUNNING
                elif ev == "CLOSE":
                    if state == _RUNNING or state == _PAUSED:
                        state = _CLOSED
                elif ev == "REOPEN":
                    if state == _CLOSED:
                        state = _RUNNING
            # Check breach after events at t
            if state != _NOT_OPENED and breached is None and used > _LIMITS[prio]:
                breached = t

        if state == _NOT_OPENED:
            continue

        # Advance to "now"
        if state == _RUNNING:
            bt = B(now)
            new_used = used + bt - b_last
            if breached is None:
                lim = _LIMITS[prio]
                if new_used > lim:
                    breached = Binv(lim - used + b_last + 1)
            used = new_used

        result.append({
            "ticket_id": tid,
            "priority": prio,
            "used_minutes": used,
            "breached": breached is not None,
            "breached_at": breached,
            "status": _STATUS_NAMES[state],
        })

    return result
